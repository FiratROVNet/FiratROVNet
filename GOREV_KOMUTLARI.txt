  # Bir ROV'un görev durumunu öğren

  filo.git(rov_id=0, x=100, y=50, z=-20)       # ROV-0'ı hedefe gönder

  filo.git(0, 1, "rov")                         # ROV-0 → ROV-1'in pozisyonuna git

  ── Çoklu Waypoint ─────────────────────────────────────────────

  # Rota listesiyle gönderin — ROV sırayla her noktaya uğrar
  filo.git(0, [(0,0), (100,0), (100,100), (0,100)], z=-20, ai=True)

  # Tüm grup-0 ROV'larını hedefe gönder (lider önce, takipçiler arkasında)

  # Yönler: ileri, geri, sağ, sol, çık (yüzey), bat (derine), dur

  # ROV-0'ın x,y konumunu koruyarak sadece derinliği değiştir

  BÖLÜM 3 — FORMASYON
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ── Kullanılabilir Formasyon İsimleri ──────────────────────────

  LINE      Tek sıra
  COLUMN    Sütun (art arda)
  V_SHAPE   V şekli (kuş uçuşu)
  WEDGE     Kama
  TRIANGLE  Üçgen
  CROSS     Haç
  SPREAD    Yayılım (geniş alan)
  STAR      Yıldız
  HEXAGON   Altıgen
  WAVE      Dalga
  SPIRAL    Spiral
  TSHAPE    T şekli

  ── Temel Kullanım ─────────────────────────────────────────────

  filo.formasyon("LINE",     aralik=15, g_id=0)
  filo.formasyon("V_SHAPE",  aralik=20, g_id=0)
  filo.formasyon("SPREAD",   aralik=40, g_id=0)  # Geniş tarama için
  filo.formasyon("TRIANGLE", aralik=15, g_id=0)

  # 3D formasyon (Z ekseninde de yayılım)
  filo.formasyon("LINE", aralik=15, is_3d=True, g_id=0)

  ── Otomatik Formasyon Seçimi ───────────────────────────────────

  # Convex hull bazlı en uygun formasyon seçilir
  filo.formasyon_sec()                           # Varsayılan grup
  filo.formasyon_sec(g_id=1)                     # Grup-1 için

  ── Pratik Örnekler ────────────────────────────────────────────

  # Grup-0'ı kama formasyonuna sok, lider hedefe gitsin, takipçiler takip etsin
  filo.formasyon("WEDGE", aralik=15, g_id=0)
  filo.change_mode(g_id=0, new_mode=1)
  filo.git(0, 100, 50, -20, ai=True)

  # İki grup farklı formasyon alsın
  filo.formasyon("V_SHAPE", aralik=15, g_id=0)
  filo.formasyon("COLUMN",  aralik=12, g_id=1)

  # Geniş alana yayıl, bağımsız hareket
  filo.formasyon("SPREAD", aralik=40, g_id=0)
  filo.change_mode(g_id=0, new_mode=0)

  BÖLÜM 4 — ALAN TARAMA GÖREVİ
  (Boustrophedon / Lawn-mower — paralel şeritlerle zigzag tarama)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ── Temel Kullanım ─────────────────────────────────────────────

  # Grup-0 ile (0,0)–(200,200) alanını varsayılan ayarlarla tara
  filo.alan_tarama_gorevi.baslat(grup_id=0, alan=(0, 0, 200, 200))

  ── Tüm Parametreler ───────────────────────────────────────────

  filo.alan_tarama_gorevi.baslat(
      grup_id            = 0,              # Hangi grup tarasın
      alan               = (0, 0, 200, 200),  # (x_min, y_min, x_max, y_max)
      derinlik           = -20.0,          # Tarama derinliği (metre, negatif)
      serit_araligi      = 15.0,           # Paralel şeritler arası mesafe (m)
      iletisim_menzili   = None,           # None → config'den GATLimitleri.KOPMA alır
      gereken_rov_sayisi = None,           # None → gruptaki tüm idle ROV'lar
      gorev_adi          = "alan_tarama",
      sessiz             = True,           # False → konsola ayrıntılı log yazar
  )

  ── Alan Formatları ────────────────────────────────────────────

  alan=(0, 0, 200, 200)                     # tuple (x_min, y_min, x_max, y_max)
  alan=(0, 0, 200, 200, -15)                # tuple 5'li: son eleman z derinliği
  alan={"x_min":0,"y_min":0,"x_max":200,"y_max":200,"z":-15}  # dict

  ── Örnekler ───────────────────────────────────────────────────

  # 1) İki grup eş zamanlı farklı bölge tarasın
  filo.alan_tarama_gorevi.baslat(grup_id=0, alan=(-150, 0, 0, 200), derinlik=-20.0, sessiz=False)
  filo.alan_tarama_gorevi.baslat(grup_id=1, alan=(0, 0, 150, 200),  derinlik=-20.0, sessiz=False)

  # 2) Sığ ve dar kanal tarama (geniş şerit = hızlı tarama)
  filo.alan_tarama_gorevi.baslat(grup_id=0, alan=(-50,-150,50,150), derinlik=-10.0, serit_araligi=10.0)

  # 3) Sadece 2 ROV ile hızlı genel tarama
  filo.alan_tarama_gorevi.baslat(grup_id=0, alan=(0,0,300,300), gereken_rov_sayisi=2, serit_araligi=30.0)

  ── Durum Sorgulama ────────────────────────────────────────────

  filo.alan_tarama_gorevi.aktif_planlar                   # Tüm aktif planlar
  filo.alan_tarama_gorevi.aktif_planlar[0].asama          # "yaklasma" veya "tarama"
  filo.alan_tarama_gorevi.aktif_planlar[0].lider_id       # Lider ROV ID'si
  filo.alan_tarama_gorevi.aktif_planlar[0].rota_by_rov    # {rov_id: [(x,y), ...]}

  ── Durdurma / Güncelleme ──────────────────────────────────────

  filo.alan_tarama_gorevi.guncelle()                # Biten ROV'ları idle'a al
  filo.alan_tarama_gorevi.guncelle(grup_id=0)       # Sadece grup-0 güncelle
  filo.alan_tarama_gorevi.durdur(grup_id=0)         # Durdur, lider takibine geç
  filo.alan_tarama_gorevi.durdur(grup_id=0, lideri_takip_et=False)  # Tamamen durdur

  BÖLÜM 5 — ARAMA KURTARMA GÖREVİ

  ── Temel Kullanım ─────────────────────────────────────────────

  # Grup-0 ile (0,0)–(200,200) alanında "person" ara
  filo.arama_kurtarma_gorevi.baslat(
      grup_id=0,
      alan=(0, 0, 200, 200),
      hedef_siniflari=["person"],
  )

  ── Tüm Parametreler ───────────────────────────────────────────

  filo.arama_kurtarma_gorevi.baslat(
      grup_id            = 0,
      alan               = (0, 0, 200, 200),
      hedef_siniflari    = ["person"],     # YOLO sınıf isimleri; None → tüm sınıflar
      model_path         = "yolov8n.pt",   # YOLO model dosyası
      derinlik           = -20.0,
      min_confidence     = 0.5,            # YOLO güven eşiği (0.0–1.0)
      gereken_rov_sayisi = None,
      sessiz             = True,
  )

  ── Örnekler ───────────────────────────────────────────────────

  # 1) Birden fazla hedef sınıfı
  filo.arama_kurtarma_gorevi.baslat(
      grup_id=0, alan=(0, 0, 400, 400),
      hedef_siniflari=["person", "boat", "diver"],
      min_confidence=0.6,
  )

  # 2) Tüm haritayı tara, yüksek güven eşiğiyle insan tespit et
  filo.arama_kurtarma_gorevi.baslat(
      grup_id=0,
      alan=(-150, -150, 150, 150),
      hedef_siniflari=["person"],
      min_confidence=0.75,
      sessiz=False,
  )

  # 3) Özel eğitilmiş model kullan
  filo.arama_kurtarma_gorevi.baslat(
      grup_id=0, alan=(0, 0, 200, 200),
      model_path="my_underwater_model.pt",
  )

  ── Tespit Sonucunu Sorgula ────────────────────────────────────

  tespit = filo.arama_kurtarma_gorevi.son_tespit
  if tespit:
      print("Bulan ROV  :", tespit.rov_id)
      print("Sınıf      :", tespit.class_name)   # "person" vb.
      print("Güven      :", tespit.confidence)   # 0.0–1.0
      print("Kamera bbox:", tespit.bbox)          # (x1, y1, x2, y2)

  filo.arama_kurtarma_gorevi.aktif_plan           # Aktif plan (None = görev yok)

  ── Durdurma / Güncelleme ──────────────────────────────────────

  filo.arama_kurtarma_gorevi.guncelle()           # YOLO kontrol + rota güncelle
  filo.arama_kurtarma_gorevi.durdur()             # Görevi durdur
  filo.arama_kurtarma_gorevi.durdur(lideri_takip_et=False)

  filo.imha_gorevi.koordinat_imha_baslat(

      hedef         = (100.0, 80.0, -15.0),   # (x, y, z)
      imha_mesafesi = 8.0,    # Bu mesafeye girince patlat (metre; varsayılan 8)
      sessiz        = True,
      hedef_entity  = None,   # Ursina Entity nesnesi (varsa patlama efekti verilir)

  sonuc = filo.imha_gorevi.guncelle()
  if sonuc and sonuc.basarili:

  filo.imha_gorevi.alan_imha_baslat(
      grup_id            = 0,
      alan               = (0, 0, 200, 200),
      hedef_siniflari    = ["mine", "bomb"],  # None → tüm sınıflar
      model_path         = "yolov8n.pt",
      derinlik           = -20.0,
      imha_mesafesi      = 8.0,
      gereken_rov_sayisi = None,
      sessiz             = True,
  )

  # Örnek: Geniş alanda mayın tara ve imha et
  filo.imha_gorevi.alan_imha_baslat(
      grup_id=0, alan=(-200,-200,200,200),
      hedef_siniflari=["mine"], imha_mesafesi=6.0, sessiz=False,

  filo.imha_gorevi.gorevli_rov_id   # İmha ROV'unun ID'si
  filo.imha_gorevi.imha_mesafesi    # Patlama tetikleme mesafesi
  filo.imha_gorevi.tespit           # Son YOLO tespiti (YoloTespit objesi veya None)

  ── Durdurma / Güncelleme ──────────────────────────────────────

  sonuc = filo.imha_gorevi.guncelle()           # İmha tamamlandıysa ImhaSonucu döner
  filo.imha_gorevi.durdur()                     # Görevi durdur
  filo.imha_gorevi.durdur(lideri_takip_et=False)

  # ROV-0'a kamera ekle

  filo.camera_manager.yolo_son_tespitler[0]    # ROV-0'ın tespitleri

  # Örnek: Grup liderlerine kamera + YOLO aç, sonra görev başlat
  for rov in filo.g_rovs.get(0, []):
      if getattr(rov, 'role', 0) == 1:
          filo.camera_manager.kamera_ekle(rov_id=rov.id)
          filo.yolo_baslat(rov.id)
  filo.arama_kurtarma_gorevi.baslat(grup_id=0, alan=(0, 0, 200, 200))
